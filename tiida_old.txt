static BLEUUID serviceUUID("0000fff0-0000-1000-8000-00805f9b34fb");
static BLEUUID    charUUID("0000fff1-0000-1000-8000-00805f9b34fb"); 
static BLEUUID   writeUUID("0000fff2-0000-1000-8000-00805f9b34fb"); 

BLERemoteCharacteristic* pNotifyChar; 
BLERemoteCharacteristic* pWriteChar;
BLEClient* pClient; 
bool connected = false; 
String obdResponse = ""; 
bool dataReceived = false;

// 🌐 ตั้งค่าระบบเครือข่ายสำหรับส่งข้อมูลเข้าเครื่องคอมพิวเตอร์/NAS (หน้า 3)

enum OBDStates { GET_RPM, GET_SPEED, GET_COOLANT, GET_VOLTS, GET_IAT, GET_STFT, GET_LTFT, GET_VACUUM, GET_LOAD, GET_O2 };
OBDStates obdState = GET_RPM;

int currentPage = 1; unsigned long lastPagePress = 0;

// ตัวแปรเก็บค่าจริงระบบเครื่องยนต์ (เริ่มต้นที่เลขศูนย์ทั้งหมด)

// 🚗 ระบบคำนวณระยะทางไมล์จำลองและการซ่อมบำรุง
float virtualOdometer = 152340.0;

float serviceGear = 180000.0;   
float servicePlug = 250000.0;

static void notifyCallback(BLERemoteCharacteristic* pBLERemoteCharacteristic, uint8_t* pData, size_t length, bool isNotify) {
  for (size_t i = 0; i < length; i++) { 
    char c = (char)pData[i];
    if (c == '>') dataReceived = true; 
    else obdResponse += c; 
  }
}

void sendOBDCommand(String cmd) { 
  if (!connected || pWriteChar == nullptr) return; 
  obdResponse = ""; dataReceived = false; cmd += "\r"; 
  pWriteChar->writeValue((uint8_t*)cmd.c_str(), cmd.length(), false); 
}

long hexToDec(String hexStr) { return strtol(hexStr.c_str(), NULL, 16); }

bool connectToOBD2() {
    BLEAddress obdAddress(target_mac); 
    pClient = BLEDevice::createClient();
    if (!pClient->connect(obdAddress)) return false;
    pClient->setMTU(512); 
    BLERemoteService* pRemoteService = pClient->getService(serviceUUID); 
    if (pRemoteService == nullptr) return false;
    pNotifyChar = pRemoteService->getCharacteristic(charUUID); 
    pWriteChar = pRemoteService->getCharacteristic(writeUUID);
    if (pNotifyChar == nullptr || pWriteChar == nullptr) return false;
    if (pNotifyChar->canNotify()) pNotifyChar->registerForNotify(notifyCallback);
    return true;
}

void drawTopTachoBarDirect(int currentRpm) {
  int startX = 12, startY = 28, barWidth = 14, barHeight = 22, gap = 4;
  int totalBars = 12; 
  int activeBars = map(currentRpm, 0, 7000, 0, totalBars);
  if(activeBars > totalBars) activeBars = totalBars;

  for (int i = 0; i < totalBars; i++) {
    uint16_t color = lcd.color565(40, 40, 40); 
    if (i < activeBars) {
      if (i < 8) color = lcd.color565(173, 255, 47);   
      else if (i < 10) color = TFT_YELLOW;             
      else color = TFT_RED;                            
    }
    int px = startX + (i * (barWidth + gap));
    for (int h = 0; h < barHeight; h++) {
      int shiftX = (barHeight - h) / 3; 
      lcd.drawFastHLine(px + shiftX, startY + h, barWidth, color);
    }
  }
}

void uploadFileToNAS() {
  uploadStatus = "WiFi Conn...";
  WiFi.begin(ssid, password);
  unsigned long startAttempt = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - startAttempt < 6000) { delay(200); }
  
  if (WiFi.status() == WL_CONNECTED) {
    uploadStatus = "Sending...";
    HTTPClient http; http.begin(nas_url);
    http.addHeader("Content-Type", "text/csv");
    String csvData = String(virtualOdometer) + "," + String(coolant) + "," + String(voltage);
    int httpResponseCode = http.POST(csvData);
    if (httpResponseCode > 0) uploadStatus = "Success NAS";
    else uploadStatus = "HTTP Error";
    http.end();
  } else { uploadStatus = "WiFi Fail"; }
  WiFi.disconnect(true);
}

void setup() {
  Serial.begin(115200);
  pinMode(21, OUTPUT); digitalWrite(21, HIGH); // สั่งเปิดไฟจอพิน 21 ทันที
  
  lcd.init(); lcd.setRotation(0); lcd.invertDisplay(false); lcd.fillScreen(TFT_BLACK);
  
  lcd.setFont(&fonts::Font2); lcd.setTextColor(TFT_WHITE, TFT_BLACK);
  lcd.drawString("NISSAN TIIDA OBD2 CLUSTER", 10, 10);
  lcd.drawString("Searching NEXPEAK BLE...", 10, 40);

  // สั่งเริ่มต้นระบบบลูทูธเพื่อล็อกการเชื่อมต่อกับตัวรถจริง
  BLEDevice::init("TIIDA_CYD_REAL");
  
  // 🔥 สั่งลูปสแกนรอสัญญาณจนกว่าจะเจอรถยนต์ (ตัดการปล่อยข้ามผ่านเข้าโหมดจำลองทิ้งถาวร)
  while (!connected) {
    if (connectToOBD2()) {
      connected = true;
      lcd.setTextColor(TFT_GREEN, TFT_BLACK); lcd.drawString("NEXPEAK Connected OK!   ", 10, 80);
      delay(1000);
      sendOBDCommand("ATZ");   delay(500); 
      sendOBDCommand("ATE0");  delay(250); 
      sendOBDCommand("ATSP0"); delay(250);
      uploadStatus = "Connected";
    } else {
      connected = false;
      lcd.setTextColor(TFT_RED, TFT_BLACK); lcd.drawString("OBD2 Offline. Retrying...", 10, 80);
      delay(1000);
    }
  }
  lcd.fillScreen(TFT_BLACK);
  lastFuelCalcTime = millis();
}

void loop() {
  // --- 1. ระบบเช็กทัชสกรีนสลับหน้าจอ (ปุ่ม NEXT ขวาล่าง) ---
  int touchX = 0, touchY = 0;
  if (lcd.getTouch(&touchX, &touchY)) {
    if (touchX >= 225 && touchX <= 320 && touchY >= 210 && touchY <= 240) {
      if (millis() - lastPagePress > 500) {
        lastPagePress = millis();
        currentPage++;
        if (currentPage > 3) currentPage = 1; 
        lcd.fillScreen(TFT_BLACK); 
      }
    }
    // ปุ่มกดส่งข้อมูล CSV to NAS (หน้า 3 ซ้ายล่าง)
    if (currentPage == 3 && touchX >= 10 && touchX <= 110 && touchY >= 210 && touchY <= 240) {
      if (millis() - lastPagePress > 800) {
        lastPagePress = millis();
        uploadFileToNAS();
      }
    }
  }

  // --- 2. ลูปยิงสลับคำสั่งวิเคราะห์ PIDs จริงจากเครื่องยนต์ Nissan Tiida 1.8L ---
  if (connected) {
    static unsigned long lastQueryTime = 0; static bool waitingForData = false;
    if (!waitingForData && (millis() - lastQueryTime > 45)) {
      lastQueryTime = millis(); waitingForData = true;
      switch (obdState) {
        case GET_RPM:     sendOBDCommand("010C"); break;
        case GET_SPEED:   sendOBDCommand("010D"); break;
        case GET_COOLANT: sendOBDCommand("0105"); break;
        case GET_VOLTS:   sendOBDCommand("ATRV"); break;
        case GET_IAT:     sendOBDCommand("010F"); break;
        case GET_STFT:    sendOBDCommand("0106"); break;
        case GET_LTFT:    sendOBDCommand("0107"); break;
        case GET_VACUUM:  sendOBDCommand("010B"); break;
        case GET_LOAD:    sendOBDCommand("0104"); break;
        case GET_O2:      sendOBDCommand("0114"); break;
      }
    }
    
    if (waitingForData && dataReceived) {
      waitingForData = false; obdResponse.replace("\r", ""); obdResponse.replace("\n", ""); obdResponse.replace(" ", "");
      if (obdResponse.indexOf("41") >= 0) {
        int idx = obdResponse.indexOf("41"); String dataBytes = obdResponse.substring(idx + 4);
        switch (obdState) {
          case GET_RPM: rpm = (hexToDec(dataBytes.substring(0, 2)) * 256.0 + hexToDec(dataBytes.substring(2, 4))) / 4.0; obdState = GET_SPEED; break;
          case GET_SPEED: currentSpeed = hexToDec(dataBytes.substring(0, 2)); obdState = GET_COOLANT; break;
          case GET_COOLANT: coolant = hexToDec(dataBytes.substring(0, 2)) - 40; obdState = GET_VOLTS; break;
          case GET_IAT: iat = hexToDec(dataBytes.substring(0, 2)) - 40; obdState = GET_STFT; break;
          case GET_STFT: stft = (hexToDec(dataBytes.substring(0, 2)) - 128) * 100.0 / 128.0; obdState = GET_LTFT; break;
          case GET_LTFT: ltft = (hexToDec(dataBytes.substring(0, 2)) - 128) * 100.0 / 128.0; obdState = GET_VACUUM; break;
          case GET_VACUUM: {
            int M_idx = obdResponse.indexOf("410B");
if (M_idx >= 0) { float mapKpa = strtol(obdResponse.substring(M_idx + 4, M_idx + 6).c_str(), NULL, 16); vacuum = (101.3 - mapKpa) * 0.2953; if (vacuum < 0.0) vacuum = 0.0; }
obdState = GET_LOAD; break;
}
case GET_LOAD: engineLoad = hexToDec(dataBytes.substring(0, 2)) * 100.0 / 255.0; obdState = GET_O2; break;
case GET_O2: o2Sensor = hexToDec(dataBytes.substring(0, 2)) / 200.0; obdState = GET_RPM; break;
}
} else if (obdState == GET_VOLTS) {
obdResponse.replace("V", ""); voltage = obdResponse.toFloat(); if(voltage == 0.0) voltage = 14.1;
obdState = GET_IAT;
} else { obdState = GET_RPM; }
}
// คำนวณระยะทางกิโลเมตรสะสมจริงอิงความเร็วรถยนต์จากกล่อง ECU
unsigned long currentTime = millis();
float timeElapsedHours = (currentTime - lastFuelCalcTime) / 3600000.0;
